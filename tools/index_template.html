#include <iostream>
#include <vector>

int main() {
    std::vector<int> v{3, 1, 4, 1, 5};
    long sum = 0;
    for (int x : v) sum += x;   // a few CPU instructions
    std::cout << "sum = " << sum << '\n';
}

$ g++ -std=c++23 -O2 main.cpp -o main && ./main
sum = 14
